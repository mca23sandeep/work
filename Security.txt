Siren Federate: Query
++++++++++++++++++++++++++++++++++++++++++++


{
  "select": ["employees.name", "employees.email", "departments.department_name", "worklogs.date", "worklogs.hours_worked"],
  "from": {
    "join": {
      "indices": [
        { "index": "employees", "alias": "employees" },
        { "index": "departments", "alias": "departments" },
        { "index": "worklogs", "alias": "worklogs" }
      ],
      "on": {
        "employees.department_id": "departments.department_id",
        "employees.employee_id": "worklogs.employee_id"
      }
    }
  }
}
