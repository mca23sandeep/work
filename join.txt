Elasticsearch plugins and settings.

employees

{
  "employee_id": "E001",
  "name": "John Doe",
  "email": "john.doe@example.com",
  "department_id": "D001"
}


departments
++++++++++++++++++++++

{
  "department_id": "D001",
  "department_name": "Sales"
}


worklogs
++++++++++++++++++++++

{
  "worklog_id": "W001",
  "employee_id": "E001",
  "date": "2025-01-20",
  "hours_worked": 8,
  "project_id": "P001"
}


Siren Federate: Query
++++++++++++++++++++++++++++++++++++++++++++


{
  "select": ["employees.name", "employees.email", "departments.department_name", "worklogs.date", "worklogs.hours_worked"],
  "from": {
    "join": {
      "indices": [
        { "index": "employees", "alias": "employees" },
        { "index": "departments", "alias": "departments" },
        { "index": "worklogs", "alias": "worklogs" }
      ],
      "on": {
        "employees.department_id": "departments.department_id",
        "employees.employee_id": "worklogs.employee_id"
      }
    }
  }
}

Query Request in Siren Federate
++++++++++++++++++++++++++++++++++++++++++++
POST /_siren/query
{
  "query": {
    "type": "join",
    "indices": ["employees", "departments"],
    "filters": [
      {
        "source": "employees",
        "destination": "departments",
        "source_field": "department_id",
        "destination_field": "department_id"
      }
    ],
    "fields": {
      "employee_id": "employees.employee_id",
      "name": "employees.name",
      "working_hours": "employees.working_hours",
      "department_name": "departments.name"
    }
  }
}

Query Request in Siren Federate 
++++++++++++++++++++++++++++++++++++++++++++
{
  "query": {
    "type": "join",
    "indices": ["employees", "departments", "worklog"],
    "filters": [
      {
        "source": "employees",
        "destination": "departments",
        "source_field": "department_id",
        "destination_field": "department_id"
      },
      {
        "source": "employees",
        "destination": "worklog",
        "source_field": "employee_id",
        "destination_field": "employee_id"
      }
    ],
    "fields": {
      "employee_id": "employees.employee_id",
      "name": "employees.name",
      "working_hours": "employees.working_hours",
      "department_name": "departments.name",
      "worklogs": {
        "worklog_id": "worklog.worklog_id",
        "date": "worklog.date",
        "hours_worked": "worklog.hours_worked"
      }
    }
  }
}
